Syntax:
* means 0 or more

program ::= statement*

statement ::= "let" VARIABLE "=" value ";" | 
                VARIABLE "=" value ";" |
                "if" truefalse "{" statement* "}" (( "elif" truefalse "{" statement* "}")* ("else" "{" statement* "}" ))? ";" | 
                "while" truefalse "{" statement* "}" ";" |
                "print" "(" value ")" ";"

truefalse ::= comparable comparison comparable 
comparable ::= NUMBER | VARIABLE
comparison ::= ">" | "=="

value ::= VARIABLE | NUMBER | value arithmetic value 
arithmetic ::= "+" | "-"

NUMBER ::= [0-9]+
VARIABLE ::= [a-zA-Z_][a-zA-Z0-9_]* forbid: "while" "if" "print" "let" "else" "elif"
